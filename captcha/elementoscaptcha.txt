<script>
function verificarCaptcha()
{
	var tmptxt = $("#tmptxt").val();
	var acton = $("#acton").val();
	$.ajax({
		type: "POST",
		url: "captcha/verifficaptcha.php",
		data: "tmptxt="+tmptxt+"&action="+acton,
		async:true,
		beforeSend: function(objeto){
			$('#captchaResult').html("Un momento, estamos verificando si eres humano");
		},
		success: function(datos){
			//alert(datos);
			//$("#result").html("seleccione un mes...");
			if(datos=='OK')
			{
				$('#captchaResult').html("Excelente!... eres humano, ahora puedes enviar el formulario");	
				$("#boton").css("display","block");
				$("#btgest").css("display","none");			
			}
			else
			{
				$('#captchaResult').html("Lo sentimos, no pareces humano, intenta de nuevo!");				
			}
		},
		timeout: 300000
	});
}
</script>
